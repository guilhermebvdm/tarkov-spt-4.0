@using System.Linq
@using Microsoft.AspNetCore.Components.Web

<div class="trl-select-container">
    @if (!string.IsNullOrEmpty(Label))
    {
        <label class="trl-label">@Label</label>
    }
    
    <div class="trl-select-wrapper">
        <select class="trl-native-select" value="@Value" @onchange="HandleChange">
            @if (string.IsNullOrEmpty(Value))
            {
                <option value="" disabled selected>@Localization.T("Select...")</option>
            }
            @if (Items != null)
            {
                @foreach (var item in Items)
                {
                    <option value="@item">@Localization.T(item)</option>
                }
            }
        </select>
        <svg class="trl-select-arrow" focusable="false" viewBox="0 0 24 24" aria-hidden="true"><path d="M7 10l5 5 5-5z"></path></svg>
    </div>
</div>

@code {
    [Parameter] public string Label { get; set; } = "";
    [Parameter] public string Value { get; set; } = "";
    [Parameter] public EventCallback<string> ValueChanged { get; set; }
    [Parameter] public List<string> Items { get; set; } = new List<string>();

    private async Task HandleChange(ChangeEventArgs e)
    {
        Value = e.Value?.ToString() ?? "";
        await ValueChanged.InvokeAsync(Value);
    }
}
