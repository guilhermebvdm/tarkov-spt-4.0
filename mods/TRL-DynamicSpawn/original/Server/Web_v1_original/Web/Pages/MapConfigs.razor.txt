@code {
    public TRLConfig Config => TRLDynamicSpawnServer.Routers.TRLRouters.CurrentConfig;
    public WaveTimerConfig GlobalTimers
    {
        get
        {
            if (!Config.MapTimers.ContainsKey("global"))
            {
                Config.MapTimers["global"] = new WaveTimerConfig();
            }
            return Config.MapTimers["global"];
        }
    }

    private Dictionary<string, string> MapList = new()
    {
        { "customs", "Customs" },
        { "factory4_day", "Factory (Day)" },
        { "factory4_night", "Factory (Night)" },
        { "interchange", "Interchange" },
        { "laboratory", "The Lab" },
        { "lighthouse", "Lighthouse" },
        { "rezervbase", "Reserve" },
        { "sandbox", "Ground Zero" },
        { "sandbox_high", "Ground Zero (High)" },
        { "shoreline", "Shoreline" },
        { "tarkovstreets", "Streets of Tarkov" },
        { "woods", "Woods" }
    };

    protected override void OnInitialized()
    {
        base.OnInitialized();
        foreach (var map in MapList.Keys)
        {
            EnsureMapConfigExists(map);
        }
    }

    private void EnsureMapConfigExists(string mapId)
    {
        if (!Config.MapConfigs.ContainsKey(mapId))
        {
            double defaultDist = (mapId == "factory4_day" || mapId == "factory4_night" || mapId == "sandbox" || mapId == "sandbox_high") ? 15.0 : 30.0;
            Config.MapConfigs[mapId] = new MapSettings { SecondsBetweenWaves = 360, SafeZoneDistance = defaultDist, SpawnBubbleDistance = 300f, EnableSpawnBubble = true, TeleportMinDistance = 100f };
        }
        else
        {
            if (Config.MapConfigs[mapId].SecondsBetweenWaves <= 0)
            {
                Config.MapConfigs[mapId].SecondsBetweenWaves = 360;
            }
            if (Config.MapConfigs[mapId].SpawnBubbleDistance <= 0f)
            {
                Config.MapConfigs[mapId].SpawnBubbleDistance = 300f;
            }
            if (Config.MapConfigs[mapId].TeleportMinDistance <= 0f)
            {
                Config.MapConfigs[mapId].TeleportMinDistance = 100f;
            }
        }
    }

    private void OnSpawnBubbleChanged(string mapKey)
    {
        if (!Config.MapConfigs[mapKey].EnableSpawnBubble)
        {
            Config.MapConfigs[mapKey].EnableDespawn = false;
        }
    }

    private string GetPresetDescription(string preset)
    {
        return preset switch
        {
            "Balanced" => "50% PMCs / 50% Scavs. The classic SPT balance.",
            "PMC War" => "70% PMCs / 30% Scavs. Prepare to fight for every inch.",
            "Scav Infestation" => "30% PMCs / 70% Scavs. A true horde of scavs on the map.",
            "Quiet Raid" => "Well-spaced spawn waves. Plenty of time to loot and explore without much pressure.",
            "Warzone" => "Extremely fast waves, short respawns, and extra chance of surprise Rogue/Raider invasions. Pure chaos.",
            "Random" => "Silently picks one of the 5 presets above at the start of each raid.",
            _ => ""
        };
    }
}
